cmake_minimum_required(VERSION 3.16)
if(COMMAND idf_component_register)
    idf_component_register(SRCS "src/frame_reader.c" "src/yamux.c" "src/aead.c" "src/crypto_psa.c" "src/handshake.c" "src/json_internal.c" "src/session.c" "src/work.c" "src/tls_mbedtls.c" "src/connect.c" "src/dns_lwip.c" "src/client.c" "src/client_port_idf.c"
        INCLUDE_DIRS "include" PRIV_REQUIRES mbedtls espressif__cjson lwip esp_timer freertos)
    target_compile_definitions(${COMPONENT_LIB} PUBLIC EFRP_HAS_TLS=1 EFRP_HAS_CONNECT=1 EFRP_HAS_CLIENT=1)
else()
    project(esp_frp VERSION 0.1.0 LANGUAGES C)
    set(CMAKE_C_STANDARD 11)
    set(CMAKE_C_STANDARD_REQUIRED ON)
    set(EFRP_PSA_SOURCE_DIR "" CACHE PATH "Optional official TF-PSA-Crypto source for production-backend host tests")
    set(EFRP_MBEDTLS_SOURCE_DIR "" CACHE PATH "Optional official Mbed TLS 4.1.0 source for TLS host tests")
    set(EFRP_TLS_SOURCE "")
    if(EFRP_MBEDTLS_SOURCE_DIR)
        if(EFRP_PSA_SOURCE_DIR)
            message(FATAL_ERROR "Mbed TLS includes its own pinned PSA source; specify only EFRP_MBEDTLS_SOURCE_DIR")
        endif()
        set(ENABLE_PROGRAMS OFF CACHE BOOL "Build upstream programs" FORCE)
        set(ENABLE_TESTING OFF CACHE BOOL "Build upstream test suites" FORCE)
        add_subdirectory("${EFRP_MBEDTLS_SOURCE_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/mbedtls")
        set(EFRP_CRYPTO_SOURCE src/crypto_psa.c)
        set(EFRP_TLS_SOURCE src/tls_mbedtls.c src/session.c src/work.c src/client.c)
        set(EFRP_CRYPTO_LIBRARY MbedTLS::mbedtls MbedTLS::mbedx509 TF-PSA-Crypto::tfpsacrypto)
    elseif(EFRP_PSA_SOURCE_DIR)
        set(ENABLE_PROGRAMS OFF CACHE BOOL "Build upstream programs" FORCE)
        set(ENABLE_TESTING OFF CACHE BOOL "Build upstream test suites" FORCE)
        add_subdirectory("${EFRP_PSA_SOURCE_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/psa")
        set(EFRP_CRYPTO_SOURCE src/crypto_psa.c)
        set(EFRP_CRYPTO_LIBRARY TF-PSA-Crypto::tfpsacrypto)
    else()
        find_package(OpenSSL 3.0 REQUIRED COMPONENTS Crypto)
        set(EFRP_CRYPTO_SOURCE src/crypto_openssl.c)
        set(EFRP_CRYPTO_LIBRARY OpenSSL::Crypto)
    endif()
    find_package(cJSON 1.7.19 EXACT CONFIG REQUIRED)
    add_library(esp_frp STATIC src/frame_reader.c src/yamux.c src/aead.c src/handshake.c src/json_internal.c ${EFRP_CRYPTO_SOURCE} ${EFRP_TLS_SOURCE})
    if(EFRP_MBEDTLS_SOURCE_DIR)
        target_compile_definitions(esp_frp PUBLIC EFRP_HAS_TLS=1)
    else()
        target_compile_definitions(esp_frp PUBLIC EFRP_HAS_TLS=0)
    endif()
    target_link_libraries(esp_frp PRIVATE ${EFRP_CRYPTO_LIBRARY} cjson)
    target_include_directories(esp_frp PUBLIC include)
    target_compile_definitions(esp_frp PUBLIC EFRP_HAS_CONNECT=0 EFRP_HAS_CLIENT=0)
    target_compile_options(esp_frp PRIVATE -Wall -Wextra -Werror -Wconversion)
    include(CTest)
    if(BUILD_TESTING)
        if(UNIX)
            find_package(Threads REQUIRED)
            add_library(connect_host_test STATIC src/connect.c tests/dns_fixture.c)
            target_include_directories(connect_host_test PUBLIC include PRIVATE src tests)
            target_compile_options(connect_host_test PRIVATE -Wall -Wextra -Werror -Wconversion -UNDEBUG)
            add_executable(connect_test tests/connect_test.c)
            target_link_libraries(connect_test PRIVATE connect_host_test)
            target_compile_options(connect_test PRIVATE -Wall -Wextra -Werror -UNDEBUG)
            add_test(NAME connect COMMAND connect_test)
            set_tests_properties(connect PROPERTIES TIMEOUT 30)
            add_executable(dns_test tests/dns_test.c src/dns_lwip.c)
            target_include_directories(dns_test PRIVATE include src tests/lwip-stub)
            target_link_libraries(dns_test PRIVATE Threads::Threads)
            target_compile_options(dns_test PRIVATE -Wall -Wextra -Werror -Wconversion -UNDEBUG)
            add_test(NAME dns_adapter COMMAND dns_test)
            set_tests_properties(dns_adapter PROPERTIES TIMEOUT 30)
        endif()
        add_executable(frame_reader_test tests/frame_reader_test.c)
        target_link_libraries(frame_reader_test PRIVATE esp_frp)
        target_compile_options(frame_reader_test PRIVATE -Wall -Wextra -Werror -UNDEBUG)
        add_test(NAME frame_reader COMMAND frame_reader_test)
        add_executable(yamux_test tests/yamux_test.c)
        target_link_libraries(yamux_test PRIVATE esp_frp)
        target_compile_options(yamux_test PRIVATE -Wall -Wextra -Werror -UNDEBUG)
        add_test(NAME yamux COMMAND yamux_test)
        add_executable(aead_test tests/aead_test.c)
        target_include_directories(aead_test PRIVATE src)
        target_link_libraries(aead_test PRIVATE esp_frp)
        target_compile_options(aead_test PRIVATE -Wall -Wextra -Werror -UNDEBUG)
        add_test(NAME aead COMMAND aead_test)
        add_executable(handshake_test tests/handshake_test.c)
        target_link_libraries(handshake_test PRIVATE esp_frp cjson)
        target_compile_options(handshake_test PRIVATE -Wall -Wextra -Werror -UNDEBUG)
        add_test(NAME handshake COMMAND handshake_test)
        if(EFRP_MBEDTLS_SOURCE_DIR AND UNIX)
            find_program(GO_EXECUTABLE go)
            if(NOT GO_EXECUTABLE)
                message(FATAL_ERROR "The TLS host test requires Go >= 1.25")
            endif()
            add_executable(tls_peer tests/tls_peer.c tests/tls_contract_test.c)
            target_link_libraries(tls_peer PRIVATE esp_frp connect_host_test MbedTLS::mbedtls TF-PSA-Crypto::tfpsacrypto)
            target_compile_options(tls_peer PRIVATE -Wall -Wextra -Werror -UNDEBUG)
            add_test(NAME tls_upstream COMMAND "${GO_EXECUTABLE}" run -mod=readonly .
                -tls-peer $<TARGET_FILE:tls_peer>)
            set_tests_properties(tls_upstream PROPERTIES
                WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/tests/crypto-interop" TIMEOUT 180)
            add_executable(session_peer tests/session_peer.c)
            target_link_libraries(session_peer PRIVATE esp_frp connect_host_test)
            target_compile_options(session_peer PRIVATE -Wall -Wextra -Werror -UNDEBUG)
            add_test(NAME session_upstream COMMAND "${GO_EXECUTABLE}" run -mod=readonly .
                -session-peer $<TARGET_FILE:session_peer>)
            set_tests_properties(session_upstream PROPERTIES
                WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/tests/crypto-interop" TIMEOUT 240)
            add_executable(work_peer tests/work_peer.c)
            target_link_libraries(work_peer PRIVATE esp_frp connect_host_test)
            target_compile_options(work_peer PRIVATE -Wall -Wextra -Werror -UNDEBUG)
            add_test(NAME work_upstream COMMAND "${GO_EXECUTABLE}" run -mod=readonly . -work-peer $<TARGET_FILE:work_peer>)
            set_tests_properties(work_upstream PROPERTIES
                WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/tests/crypto-interop" TIMEOUT 240)
            add_test(NAME work_faults_upstream COMMAND "${GO_EXECUTABLE}" run -mod=readonly . -work-peer $<TARGET_FILE:work_peer> -work-faults)
            set_tests_properties(work_faults_upstream PROPERTIES
                WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/tests/crypto-interop" TIMEOUT 180)
            add_executable(client_peer tests/client_peer.c tests/client_port_posix.c)
            target_include_directories(client_peer PRIVATE src tests)
            target_link_libraries(client_peer PRIVATE esp_frp connect_host_test Threads::Threads)
            target_compile_options(client_peer PRIVATE -Wall -Wextra -Werror -UNDEBUG)
            add_test(NAME client_upstream COMMAND "${GO_EXECUTABLE}" run -mod=readonly . -client-peer $<TARGET_FILE:client_peer>)
            set_tests_properties(client_upstream PROPERTIES
                WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/tests/crypto-interop" TIMEOUT 240)
        endif()
        option(EFRP_TEST_UPSTREAM_CRYPTO "Run AEAD interop with official FRP v0.71.0" OFF)
        if(EFRP_TEST_UPSTREAM_CRYPTO)
            find_program(GO_EXECUTABLE go)
            if(NOT GO_EXECUTABLE)
                message(FATAL_ERROR "The optional upstream crypto test requires Go >= 1.25")
            endif()
            add_executable(aead_peer tests/aead_peer.c)
            target_link_libraries(aead_peer PRIVATE esp_frp)
            target_compile_options(aead_peer PRIVATE -Wall -Wextra -Werror -UNDEBUG)
            add_test(NAME aead_upstream COMMAND "${GO_EXECUTABLE}" run -mod=readonly .
                -peer $<TARGET_FILE:aead_peer>)
            set_tests_properties(aead_upstream PROPERTIES
                WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/tests/crypto-interop" TIMEOUT 120)
            add_executable(handshake_peer tests/handshake_peer.c)
            target_link_libraries(handshake_peer PRIVATE esp_frp)
            target_compile_options(handshake_peer PRIVATE -Wall -Wextra -Werror -UNDEBUG)
            add_test(NAME handshake_upstream COMMAND "${GO_EXECUTABLE}" run -mod=readonly .
                -handshake-peer $<TARGET_FILE:handshake_peer>)
            set_tests_properties(handshake_upstream PROPERTIES
                WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/tests/crypto-interop" TIMEOUT 120)
        endif()
        option(EFRP_TEST_UPSTREAM_YAMUX "Run TCP interop with the FRP-pinned Go Yamux dependency" OFF)
        if(EFRP_TEST_UPSTREAM_YAMUX)
            if(NOT UNIX)
                message(FATAL_ERROR "The optional Yamux TCP peer requires a POSIX host")
            endif()
            find_program(GO_EXECUTABLE go)
            if(NOT GO_EXECUTABLE)
                message(FATAL_ERROR "The optional upstream Yamux test requires Go >= 1.23")
            endif()
            add_executable(yamux_peer tests/yamux_peer.c)
            target_link_libraries(yamux_peer PRIVATE esp_frp)
            target_compile_options(yamux_peer PRIVATE -Wall -Wextra -Werror -UNDEBUG)
            add_test(NAME yamux_upstream COMMAND "${GO_EXECUTABLE}" run -mod=readonly .
                -client $<TARGET_FILE:yamux_peer> -rounds 100)
            set_tests_properties(yamux_upstream PROPERTIES
                WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/tests/interop" TIMEOUT 120)
        endif()
    endif()
endif()
