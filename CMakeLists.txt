cmake_minimum_required(VERSION 3.16)
if(COMMAND idf_component_register)
    idf_component_register(SRCS "src/frame_reader.c" "src/yamux.c" INCLUDE_DIRS "include")
else()
    project(esp_frp VERSION 0.1.0 LANGUAGES C)
    set(CMAKE_C_STANDARD 11)
    set(CMAKE_C_STANDARD_REQUIRED ON)
    add_library(esp_frp STATIC src/frame_reader.c src/yamux.c)
    target_include_directories(esp_frp PUBLIC include)
    target_compile_options(esp_frp PRIVATE -Wall -Wextra -Werror -Wconversion)
    include(CTest)
    if(BUILD_TESTING)
        add_executable(frame_reader_test tests/frame_reader_test.c)
        target_link_libraries(frame_reader_test PRIVATE esp_frp)
        target_compile_options(frame_reader_test PRIVATE -Wall -Wextra -Werror -UNDEBUG)
        add_test(NAME frame_reader COMMAND frame_reader_test)
        add_executable(yamux_test tests/yamux_test.c)
        target_link_libraries(yamux_test PRIVATE esp_frp)
        target_compile_options(yamux_test PRIVATE -Wall -Wextra -Werror -UNDEBUG)
        add_test(NAME yamux COMMAND yamux_test)
        option(EFRP_TEST_UPSTREAM_YAMUX "Run TCP interop with the FRP-pinned Go Yamux dependency" OFF)
        if(EFRP_TEST_UPSTREAM_YAMUX)
            find_program(GO_EXECUTABLE go REQUIRED)
            add_executable(yamux_peer tests/yamux_peer.c)
            target_link_libraries(yamux_peer PRIVATE esp_frp)
            target_compile_options(yamux_peer PRIVATE -Wall -Wextra -Werror -UNDEBUG)
            add_test(NAME yamux_upstream COMMAND "${GO_EXECUTABLE}" run -mod=readonly .
                -client $<TARGET_FILE:yamux_peer> -rounds 100)
            set_tests_properties(yamux_upstream PROPERTIES
                WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/tests/interop" TIMEOUT 120)
        endif()
    endif()
endif()
