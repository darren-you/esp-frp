cmake_minimum_required(VERSION 3.20)
project(esp_frp_quic_prototype LANGUAGES C)
set(efrp_source "${CMAKE_CURRENT_SOURCE_DIR}/../../src")
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(EFRP_NGTCP2_SOURCE_DIR "" CACHE PATH "Clean pinned ngtcp2 checkout")
set(EFRP_PICOTLS_SOURCE_DIR "" CACHE PATH "Clean pinned Picotls checkout")
set(EFRP_MBEDTLS_SOURCE_DIR "" CACHE PATH "Verified official Mbed TLS 4.1.0 full release")
include("${CMAKE_CURRENT_SOURCE_DIR}/../../tools/quic_source_guard.cmake")
if(NOT EXISTS "${EFRP_MBEDTLS_SOURCE_DIR}/CMakeLists.txt")
    message(FATAL_ERROR "Specify verified full Mbed TLS 4.1.0 release source")
endif()
set(ENABLE_TESTING OFF CACHE BOOL "" FORCE)
set(ENABLE_PROGRAMS OFF CACHE BOOL "" FORCE)
set(GEN_FILES OFF CACHE BOOL "" FORCE)
add_subdirectory("${EFRP_MBEDTLS_SOURCE_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/mbedtls")
set(ENABLE_LIB_ONLY ON CACHE BOOL "" FORCE)
set(ENABLE_OPENSSL OFF CACHE BOOL "" FORCE)
set(ENABLE_PICOTLS OFF CACHE BOOL "" FORCE)
set(ENABLE_SHARED_LIB OFF CACHE BOOL "" FORCE)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
add_subdirectory("${EFRP_NGTCP2_SOURCE_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/ngtcp2")
set(picotls "${EFRP_PICOTLS_SOURCE_DIR}")
include("${CMAKE_CURRENT_SOURCE_DIR}/../../tools/quic_picotls_sources.cmake")
add_library(quic_picotls STATIC ${minicrypto_sources} "${efrp_source}/quic_random.c")
target_include_directories(quic_picotls SYSTEM PUBLIC "${picotls}/include")
target_include_directories(quic_picotls PRIVATE
    "${picotls}/deps/cifra/src" "${picotls}/deps/cifra/src/ext" "${picotls}/deps/micro-ecc")
target_compile_definitions(quic_picotls PUBLIC PTLS_HAVE_LOG=0)
target_link_libraries(quic_picotls PUBLIC TF-PSA-Crypto::tfpsacrypto)
add_library(quic_crypto STATIC "${efrp_source}/quic_crypto.c" "${EFRP_NGTCP2_SOURCE_DIR}/crypto/shared.c" "${efrp_source}/quic_certificate.c")
target_include_directories(quic_crypto PUBLIC "${efrp_source}"
    "${EFRP_NGTCP2_SOURCE_DIR}/crypto/includes" PRIVATE
    "${EFRP_NGTCP2_SOURCE_DIR}/lib" "${EFRP_NGTCP2_SOURCE_DIR}/crypto"
    "${CMAKE_CURRENT_BINARY_DIR}/ngtcp2")
target_compile_definitions(quic_crypto PRIVATE HAVE_CONFIG_H NGTCP2_STATICLIB)
target_compile_options(quic_crypto PRIVATE -Wall -Wextra -Werror)
target_link_libraries(quic_crypto PUBLIC ngtcp2_static quic_picotls MbedTLS::mbedx509 TF-PSA-Crypto::tfpsacrypto)
add_executable(quic_client client.c)
target_compile_definitions(quic_client PRIVATE EFRP_QUIC_HOST_MAIN)
target_compile_options(quic_client PRIVATE -Wall -Wextra -Werror)
target_link_libraries(quic_client PRIVATE quic_crypto)
add_library(quic_backend STATIC "${efrp_source}/quic.c" "${efrp_source}/stream.c" "${efrp_source}/quic_peer_security.c" "${efrp_source}/quic_peer.c")
target_include_directories(quic_backend PUBLIC "${efrp_source}" "${CMAKE_CURRENT_SOURCE_DIR}/../../include")
target_compile_options(quic_backend PRIVATE -Wall -Wextra -Werror -Wconversion)
target_link_libraries(quic_backend PUBLIC quic_crypto)
add_executable(quic_transport_peer "${CMAKE_CURRENT_SOURCE_DIR}/../quic_transport_peer.c"
    "${CMAKE_CURRENT_SOURCE_DIR}/../quic_dns_fixture.c" "${CMAKE_CURRENT_SOURCE_DIR}/../quic_socket_fixture.c"
    "${efrp_source}/aead.c" "${efrp_source}/crypto_psa.c")
target_compile_options(quic_transport_peer PRIVATE -Wall -Wextra -Werror -Wconversion)
target_include_directories(quic_transport_peer PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/..")
target_link_libraries(quic_transport_peer PRIVATE quic_backend)
add_executable(stream_yamux_test "${CMAKE_CURRENT_SOURCE_DIR}/../stream_yamux_test.c"
    "${efrp_source}/stream_yamux.c" "${efrp_source}/stream.c" "${efrp_source}/yamux.c"
    "${efrp_source}/aead.c" "${efrp_source}/crypto_psa.c")
target_compile_options(stream_yamux_test PRIVATE -Wall -Wextra -Werror -Wconversion)
target_include_directories(stream_yamux_test PRIVATE "${efrp_source}" "${CMAKE_CURRENT_SOURCE_DIR}/../../include")
target_link_libraries(stream_yamux_test PRIVATE quic_crypto)
add_executable(quic_peer_security_test "${CMAKE_CURRENT_SOURCE_DIR}/../quic_peer_security_test.c"
    "${efrp_source}/aead.c" "${efrp_source}/crypto_psa.c")
target_compile_options(quic_peer_security_test PRIVATE -Wall -Wextra -Werror -Wconversion)
target_link_libraries(quic_peer_security_test PRIVATE quic_backend)
add_executable(quic_peer_transport_peer "${CMAKE_CURRENT_SOURCE_DIR}/../quic_peer_transport_peer.c"
    "${CMAKE_CURRENT_SOURCE_DIR}/../quic_dns_fixture.c" "${efrp_source}/aead.c" "${efrp_source}/crypto_psa.c")
target_compile_options(quic_peer_transport_peer PRIVATE -Wall -Wextra -Werror -Wconversion)
target_include_directories(quic_peer_transport_peer PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/..")
target_link_libraries(quic_peer_transport_peer PRIVATE quic_backend)
