package proxy

import (
	"context"
	"crypto/tls"
	"crypto/x509"
	"io"
	"net"
	"testing"
	"time"

	"github.com/fatedier/frp/client/visitor"
	v1 "github.com/fatedier/frp/pkg/config/v1"
	"github.com/fatedier/frp/pkg/msg"
	"github.com/quic-go/quic-go"
)

func TestESPFRPFixedOfficialProviderAcceptsUnrelatedUDPSource(t *testing.T) {
	udp := func() *net.UDPConn {
		c, err := net.ListenUDP("udp4", &net.UDPAddr{IP: net.IPv4(127, 0, 0, 1)})
		if err != nil {
			t.Fatal(err)
		}
		t.Cleanup(func() { _ = c.Close() })
		return c
	}
	backend, err := net.Listen("tcp4", "127.0.0.1:0")
	if err != nil {
		t.Fatal(err)
	}
	t.Cleanup(func() { _ = backend.Close() })
	ctx, cancel := context.WithTimeout(context.Background(), 8*time.Second)
	t.Cleanup(cancel)
	const body = "public-unrelated-peer-no-proxy-or-sid"
	backendResult := make(chan error, 1)
	go func() {
		c, err := backend.Accept()
		if err != nil {
			backendResult <- err
			return
		}
		defer c.Close()
		_ = c.SetDeadline(time.Now().Add(6 * time.Second))
		got := make([]byte, len(body))
		_, err = io.ReadFull(c, got)
		if err == nil && string(got) != body {
			err = io.ErrUnexpectedEOF
		}
		if err == nil {
			_, err = c.Write([]byte("backend-received"))
		}
		if err == nil {
			ack := make([]byte, 4)
			_, err = io.ReadFull(c, ack)
			if err == nil && string(ack) != "done" {
				err = io.ErrUnexpectedEOF
			}
		}
		backendResult <- err
	}()
	common := &v1.ClientCommonConfig{}
	common.Transport.QUIC = v1.QUICOptions{MaxIdleTimeout: 5, MaxIncomingStreams: 2, KeepalivePeriod: 1}
	cfg := &v1.XTCPProxyConfig{
		ProxyBaseConfig: v1.ProxyBaseConfig{Name: "private", Type: "xtcp", ProxyBackend: v1.ProxyBackend{
			LocalIP: "127.0.0.1", LocalPort: backend.Addr().(*net.TCPAddr).Port,
		}}, Secretkey: "public-provider-secret-not-given-to-peer",
	}
	provider := NewProxy(ctx, cfg, common, nil, nil, nil, "").(*XTCPProxy)
	socket, intendedPeer := udp(), udp()
	finished := make(chan struct{})
	go func() {
		defer close(finished)
		provider.listenByQUIC(socket, intendedPeer.LocalAddr().(*net.UDPAddr), &msg.StartWorkConn{})
	}()
	t.Cleanup(func() {
		cancel()
		_ = socket.Close()
		select {
		case <-finished:
		case <-time.After(2 * time.Second):
			t.Error("official provider listener did not release")
		}
	})
	// A strict CA-based QUIC client cannot trust this official dynamic certificate.
	// This is a negative baseline, not a fixture exemption from TLS verification.
	strictCtx, strictCancel := context.WithTimeout(ctx, 2*time.Second)
	_, strictErr := quic.DialAddr(strictCtx, socket.LocalAddr().String(), &tls.Config{
		RootCAs: x509.NewCertPool(), ServerName: "127.0.0.1", NextProtos: []string{"frp"}, MinVersion: tls.VersionTLS13,
	}, &quic.Config{HandshakeIdleTimeout: time.Second})
	strictCancel()
	if strictErr == nil {
		t.Fatal("strict empty CA trust unexpectedly accepted dynamic official provider certificate")
	}
	peerSocket := udp()
	if peerSocket.LocalAddr().String() == intendedPeer.LocalAddr().String() {
		t.Fatal("fixture source collision")
	}
	session := visitor.NewQUICTunnelSession(common)
	t.Cleanup(session.Close)
	// Call the fixed official implementation. The fixture does not set skip-verify.
	if err := session.Init(peerSocket, socket.LocalAddr().(*net.UDPAddr)); err != nil {
		t.Fatal(err)
	}
	c, err := session.OpenConn(ctx)
	if err != nil {
		t.Fatal(err)
	}
	t.Cleanup(func() { _ = c.Close() })
	_ = c.SetDeadline(time.Now().Add(5 * time.Second))
	if _, err = c.Write([]byte(body)); err != nil {
		t.Fatal(err)
	}
	ack := make([]byte, len("backend-received"))
	if _, err = io.ReadFull(c, ack); err != nil || string(ack) != "backend-received" {
		t.Fatalf("backend reply %q %v", ack, err)
	}
	if _, err = c.Write([]byte("done")); err != nil {
		t.Fatal(err)
	}
	if err = <-backendResult; err != nil {
		t.Fatal(err)
	}
	t.Logf("strict X509 client rejected dynamic certificate (%v); actual official provider reached backend from %s although intended peer was %s, without SID or secret admission", strictErr, peerSocket.LocalAddr(), intendedPeer.LocalAddr())
}
